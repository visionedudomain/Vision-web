rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isAdmin() {
      return request.auth != null
        && request.auth.token.email == "visionedudomain@gmail.com";
    }

    function isApprovedStudent() {
      return request.auth != null
        && exists(/databases/$(database)/documents/students/$(request.auth.uid))
        && get(/databases/$(database)/documents/students/$(request.auth.uid)).data.status == "approved";
    }

    function hasStudentRecord() {
      return request.auth != null
        && exists(/databases/$(database)/documents/students/$(request.auth.uid));
    }

    function hasOwnAttempt(testId) {
      return request.auth != null
        && exists(/databases/$(database)/documents/attempts/$(request.auth.uid + "__" + testId));
    }

    function ownAttempt(testId) {
      return get(/databases/$(database)/documents/attempts/$(request.auth.uid + "__" + testId));
    }

    function isRewriteRequestUpdate() {
      return hasStudentRecord()
        && resource.data.studentId == request.auth.uid
        && request.resource.data.studentId == request.auth.uid
        && request.resource.data.testId == resource.data.testId
        && resource.data.status != "started"
        && request.resource.data.status == resource.data.status
        && !(request.resource.data.rewriteRequestStatus in ["approved"])
        && !(resource.data.rewriteRequestStatus in ["pending", "approved"])
        && request.resource.data.rewriteRequestStatus == "pending"
        && request.resource.data.rewriteRequestedAt is string
        && request.resource.data.rewriteRequestedAtMs is int
        && request.resource.data.rewriteReviewedAt == ""
        && request.resource.data.rewriteReviewedAtMs == 0
        && request.resource.data.rewriteReviewedBy == ""
        && request.resource.data.updatedAt is string
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          "rewriteRequestStatus",
          "rewriteRequestedAt",
          "rewriteRequestedAtMs",
          "rewriteReviewedAt",
          "rewriteReviewedAtMs",
          "rewriteReviewedBy",
          "updatedAt"
        ]);
    }

    function isAdminRewriteReviewUpdate() {
      return isAdmin()
        && request.resource.data.studentId == resource.data.studentId
        && request.resource.data.testId == resource.data.testId
        && request.resource.data.status == resource.data.status
        && request.resource.data.startedAt == resource.data.startedAt
        && request.resource.data.startedAtMs == resource.data.startedAtMs
        && request.resource.data.expiresAt == resource.data.expiresAt
        && request.resource.data.expiresAtMs == resource.data.expiresAtMs
        && request.resource.data.submittedAt == resource.data.submittedAt
        && request.resource.data.submittedAtMs == resource.data.submittedAtMs
        && resource.data.rewriteRequestStatus == "pending"
        && request.resource.data.rewriteRequestStatus in ["approved", "rejected"]
        && request.resource.data.rewriteRequestedAt == resource.data.rewriteRequestedAt
        && request.resource.data.rewriteRequestedAtMs == resource.data.rewriteRequestedAtMs
        && request.resource.data.rewriteReviewedAt is string
        && request.resource.data.rewriteReviewedAtMs is int
        && request.resource.data.rewriteReviewedBy is string
        && request.resource.data.updatedAt is string
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          "rewriteRequestStatus",
          "rewriteReviewedAt",
          "rewriteReviewedAtMs",
          "rewriteReviewedBy",
          "updatedAt"
        ]);
    }

    function isApprovedRewriteRestartUpdate(attemptId) {
      return hasStudentRecord()
        && resource.data.studentId == request.auth.uid
        && request.resource.data.studentId == request.auth.uid
        && request.resource.data.testId == resource.data.testId
        && attemptId == request.auth.uid + "__" + resource.data.testId
        && exists(/databases/$(database)/documents/published_tests/$(resource.data.testId))
        && get(/databases/$(database)/documents/published_tests/$(resource.data.testId)).data.isActive == true
        && request.resource.data.studentDisplayName
            == get(/databases/$(database)/documents/students/$(request.auth.uid)).data.displayName
        && request.resource.data.studentLoginName
            == get(/databases/$(database)/documents/students/$(request.auth.uid)).data.loginName
        && request.resource.data.testTitle
            == get(/databases/$(database)/documents/published_tests/$(resource.data.testId)).data.title
        && request.resource.data.language
            == get(/databases/$(database)/documents/published_tests/$(resource.data.testId)).data.language
        && resource.data.rewriteRequestStatus == "approved"
        && request.resource.data.status == "started"
        && request.resource.data.startedAt is string
        && request.resource.data.startedAtMs is int
        && request.resource.data.expiresAt is string
        && request.resource.data.expiresAtMs is int
        && request.resource.data.expiresAtMs > request.resource.data.startedAtMs
        && request.resource.data.submittedAt == ""
        && request.resource.data.submittedAtMs == 0
        && request.resource.data.answers is map
        && request.resource.data.score == 0
        && request.resource.data.correctCount == 0
        && request.resource.data.answeredCount == 0
        && request.resource.data.totalQuestions
            == get(/databases/$(database)/documents/published_tests/$(resource.data.testId)).data.questionCount
        && request.resource.data.percentage == 0
        && request.resource.data.attemptedAccuracy == 0
        && request.resource.data.unansweredCount == request.resource.data.totalQuestions
        && request.resource.data.performanceStatusCode == ""
        && request.resource.data.suggestionCodes.size() == 0
        && request.resource.data.rewriteRequestStatus == ""
        && request.resource.data.rewriteRequestedAt == ""
        && request.resource.data.rewriteRequestedAtMs == 0
        && request.resource.data.rewriteReviewedAt == ""
        && request.resource.data.rewriteReviewedAtMs == 0
        && request.resource.data.rewriteReviewedBy == ""
        && request.resource.data.updatedAt is string
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          "studentDisplayName",
          "studentLoginName",
          "testTitle",
          "language",
          "startedAt",
          "startedAtMs",
          "expiresAt",
          "expiresAtMs",
          "submittedAt",
          "submittedAtMs",
          "status",
          "answers",
          "score",
          "correctCount",
          "answeredCount",
          "totalQuestions",
          "percentage",
          "attemptedAccuracy",
          "unansweredCount",
          "performanceStatusCode",
          "suggestionCodes",
          "rewriteRequestStatus",
          "rewriteRequestedAt",
          "rewriteRequestedAtMs",
          "rewriteReviewedAt",
          "rewriteReviewedAtMs",
          "rewriteReviewedBy",
          "updatedAt"
        ]);
    }

    match /site/{document} {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /news/{document} {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /applications/{document} {
      allow create: if true;
      allow read, update, delete: if isAdmin();
    }

    match /test_registrations/{registrationId} {
      allow create: if request.auth == null
        && registrationId == request.resource.data.loginNameNormalized
        && request.resource.data.status == "pending"
        && request.resource.data.displayName is string
        && request.resource.data.loginName is string
        && request.resource.data.loginNameNormalized is string
        && request.resource.data.authUid is string
        && request.resource.data.authEmail is string
        && request.resource.data.mobile is string
        && request.resource.data.language in ["en", "ta"];
      allow read, update, delete: if isAdmin();
    }

    match /student_login_index/{loginNameNormalized} {
      allow get: if true;
      allow list: if isAdmin();
      allow write: if isAdmin();
    }

    match /students/{studentId} {
      allow read: if isAdmin() || (request.auth != null && request.auth.uid == studentId);
      allow write: if isAdmin();
    }

    match /tests/{testId} {
      allow read, write: if isAdmin();
    }

    match /published_tests/{testId} {
      allow read: if isAdmin()
        || (isApprovedStudent() && (resource.data.isActive == true || hasOwnAttempt(testId)));
      allow write: if isAdmin();
    }

    match /test_answer_keys/{testId} {
      allow read: if isAdmin()
        || (isApprovedStudent() && hasOwnAttempt(testId));
      allow write: if isAdmin();
    }

    match /attempts/{attemptId} {
      allow read: if isAdmin()
        || (request.auth != null && resource.data.studentId == request.auth.uid);

      allow create: if isApprovedStudent()
        && request.resource.data.studentId == request.auth.uid
        && attemptId == request.auth.uid + "__" + request.resource.data.testId
        && request.resource.data.status == "started"
        && request.resource.data.startedAtMs is int
        && request.resource.data.expiresAtMs is int
        && request.resource.data.expiresAtMs > request.resource.data.startedAtMs
        && request.resource.data.submittedAtMs == 0
        && exists(/databases/$(database)/documents/published_tests/$(request.resource.data.testId))
        && request.resource.data.totalQuestions
            == get(/databases/$(database)/documents/published_tests/$(request.resource.data.testId)).data.questionCount
        && get(/databases/$(database)/documents/published_tests/$(request.resource.data.testId)).data.isActive == true;

      allow update: if isAdminRewriteReviewUpdate()
        || isApprovedRewriteRestartUpdate(attemptId)
        || (
          isApprovedStudent()
          && resource.data.studentId == request.auth.uid
          && request.resource.data.studentId == request.auth.uid
          && request.resource.data.testId == resource.data.testId
          && request.resource.data.studentDisplayName == resource.data.studentDisplayName
          && request.resource.data.studentLoginName == resource.data.studentLoginName
          && request.resource.data.testTitle == resource.data.testTitle
          && request.resource.data.language == resource.data.language
          && request.resource.data.totalQuestions == resource.data.totalQuestions
          && resource.data.status == "started"
          && request.resource.data.status in ["submitted", "auto_submitted"]
          && request.resource.data.startedAt == resource.data.startedAt
          && request.resource.data.startedAtMs == resource.data.startedAtMs
          && request.resource.data.expiresAt == resource.data.expiresAt
          && request.resource.data.expiresAtMs == resource.data.expiresAtMs
          && request.resource.data.submittedAtMs is int
          && request.resource.data.submittedAtMs >= resource.data.startedAtMs
        )
        || isRewriteRequestUpdate();

      allow delete: if isAdmin();
    }
  }
}
